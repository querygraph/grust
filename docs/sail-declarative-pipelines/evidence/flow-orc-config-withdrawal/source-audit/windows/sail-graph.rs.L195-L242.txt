        }

            x.name == target

        });
        let (name, target) = if private {
            (name, target)
        } else {
            (
                self.definition.qualify(&name)?,
                self.definition.qualify(&target)?,
            )
        };

            pc::define_flow::Details::RelationFlowDetails(details) => {

                    spec::Plan::Query(query) => query,
                    spec::Plan::Command(_) => {
                        return Err(analysis(

                    }
