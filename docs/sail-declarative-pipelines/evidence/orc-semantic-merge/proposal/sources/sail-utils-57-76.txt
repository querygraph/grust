57: /// Merges per-file inferred schemas, normalizing each one before the merge.

60: /// happen before the merge rather than after it. A directory holding the same column as
61: /// `timestamp[ms]` in one file and `timestamp[us]` in another would otherwise fail to merge,

63: /// before [`rewrite_unsupported_fields`] ever ran.

66:     nanoseconds: NanosecondTimestamps,
67: ) -> Result<Schema> {
68:     Ok(Schema::try_merge(schemas.into_iter().map(|schema| {
69:         normalize_unsupported_fields(&schema, nanoseconds)
70:     }))?)
71: }
