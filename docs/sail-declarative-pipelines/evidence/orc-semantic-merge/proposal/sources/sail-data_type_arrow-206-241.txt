206:         let kind = match data_type {
207:             adt::DataType::Null => Kind::Null(sdt::Null::default()),
208:             adt::DataType::Binary
209:             | adt::DataType::FixedSizeBinary(_)
210:             | adt::DataType::LargeBinary
211:             | adt::DataType::BinaryView => Kind::Binary(sdt::Binary::default()),
212:             adt::DataType::Boolean => Kind::Boolean(sdt::Boolean::default()),
213:             // TODO: cast unsigned integer types to signed integer types in the query output,
214:             //   and return an error if unsigned integer types are found here.
215:             adt::DataType::UInt8 | adt::DataType::Int8 => Kind::Byte(sdt::Byte::default()),
216:             adt::DataType::UInt16 | adt::DataType::Int16 => Kind::Short(sdt::Short::default()),
217:             adt::DataType::UInt32 | adt::DataType::Int32 => Kind::Integer(sdt::Integer::default()),
218:             adt::DataType::UInt64 | adt::DataType::Int64 => Kind::Long(sdt::Long::default()),
219:             adt::DataType::Float16 => return Err(error(&data_type)),
220:             adt::DataType::Float32 => Kind::Float(sdt::Float::default()),
221:             adt::DataType::Float64 => Kind::Double(sdt::Double::default()),
222:             adt::DataType::Decimal128(precision, scale)
223:             | adt::DataType::Decimal256(precision, scale) => Kind::Decimal(sdt::Decimal {
224:                 scale: Some(scale as i32),
225:                 precision: Some(precision as i32),
226:                 type_variation_reference: 0,

228:             // FIXME: This mapping might not always be correct due to converting to Arrow data types and back.

233:             }
234:             adt::DataType::Date32 => Kind::Date(sdt::Date::default()),
235:             adt::DataType::Date64 => return Err(error(&data_type)),
