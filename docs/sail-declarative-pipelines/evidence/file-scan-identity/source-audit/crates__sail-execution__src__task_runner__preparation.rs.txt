55:         proto: &PhysicalPlanNode,

68:         debug!(
69:             "{} execution plan\n{}",
70:             TaskKeyDisplay(key),
71:             DisplayableExecutionPlan::new(plan.as_ref()).indent(true)
72:         );
73:         let plan = trace_execution_plan(

92:     fn rewrite_file_scans(
93:         &self,
94:         plan: Arc<dyn ExecutionPlan>,
95:     ) -> ExecutionResult<Arc<dyn ExecutionPlan>> {
96:         let result = plan.transform(|node| {
97:             if let Some(ds) = node.downcast_ref::<DataSourceExec>()
98:                 && let Some(base_config) = ds.data_source().downcast_ref::<FileScanConfig>()
99:             {
100:                 // DataFusion file scans can use process-local sibling state to let
101:                 // partitions steal work from a shared queue of all file groups. In Sail
102:                 // cluster mode each partition runs as an isolated task with its own
103:                 // deserialized plan, so that queue would be recreated in every task and
104:                 // every task would scan every file. Preserve-order disables sibling
105:                 // work sharing and keeps each task on its own file group.
106:                 let mut builder =
107:                     FileScanConfigBuilder::from(base_config.clone()).with_preserve_order(true);
108:                 if ds.downcast_to_file_source::<ParquetSource>().is_some()
109:                     && base_config.expr_adapter_factory.is_none()
110:                 {
111:                     let adapter_factory: Arc<dyn PhysicalExprAdapterFactory> =
112:                         Arc::new(SchemaEvolutionPhysicalExprAdapterFactory {});
113:                     builder = builder.with_expr_adapter(Some(adapter_factory));
114:                 }
115:                 return Ok(Transformed::yes(
116:                     DataSourceExec::from_data_source(builder.build()) as Arc<dyn ExecutionPlan>,
117:                 ));
118:             }
119:             Ok(Transformed::no(node))
120:         });
121:         Ok(result.data()?)
